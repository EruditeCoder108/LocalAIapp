cmake_minimum_required(VERSION 3.22.1)
project(ember_phonon LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(EMBER_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
add_library(emberphonon SHARED phonon.cpp)
target_include_directories(emberphonon PRIVATE "${EMBER_ROOT}/vendor/phonon-android/phonon2-android-arm64-v8a-0.2.11" "${EMBER_ROOT}/vendor/liquid.cpp/vendor")
target_compile_options(emberphonon PRIVATE -O3 -fvisibility=hidden "-ffile-prefix-map=${EMBER_ROOT}=.")
target_link_libraries(emberphonon PRIVATE android log dl)
target_link_options(emberphonon PRIVATE "-Wl,--version-script=${EMBER_ROOT}/app/src/main/cpp/jni-exports.map" "-Wl,-z,max-page-size=16384")

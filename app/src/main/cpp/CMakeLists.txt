cmake_minimum_required(VERSION 3.22.1)
project(bonsai_pocket LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8.2-a+dotprod" CACHE STRING "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
# Keep local workstation paths out of native diagnostic strings and debug info.
get_filename_component(EMBER_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
add_compile_options("-ffile-prefix-map=${EMBER_ROOT}=." "-fdebug-prefix-map=${EMBER_ROOT}=.")
if(DEFINED ENV{USERPROFILE})
    file(TO_CMAKE_PATH "$ENV{USERPROFILE}" EMBER_HOME)
    add_compile_options("-ffile-prefix-map=${EMBER_HOME}=/home/build" "-fdebug-prefix-map=${EMBER_HOME}=/home/build")
endif()
add_subdirectory(../../../../vendor/llama.cpp llama-build)
add_library(bonsai SHARED bonsai.cpp)
target_link_libraries(bonsai PRIVATE llama llama-common mtmd android log)
target_link_options(bonsai PRIVATE "-Wl,-z,max-page-size=16384")
